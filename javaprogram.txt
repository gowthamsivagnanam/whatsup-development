2.WAP print even number range 50 to 100 using for loop.

public class EvenNumbers {
    public static void main(String[] args) {
        for (int i = 50; i <= 100; i++) {
            if (i % 2 == 0) {
                System.out.println(i);
            }
        }
    }
}

9. WAP to Print
        a b c d
        e f g h
        i  j  k  l
       m n o p
public class AlphabetPattern {
    public static void main(String[] args) {
        char ch = 'a'; 
        int n = 4;
        for (int i = 1; i <= n; i++) { 
            for (int j = 1; j <= n; j++) { 
                System.out.print(ch + " ");
                ch++; 
            }
            System.out.println();
        }
    }
}
